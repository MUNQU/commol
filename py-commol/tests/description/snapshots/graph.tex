\section*{Dynamics}
\subsection*{Compact form}
\subsubsection*{Bin Transitions}
\noindent\texttt{pair (A, B -\textgreater{} C)}
\begin{align*}
A_{g1} \to C_{g1} &: k1 \cdot A \cdot B \\
B_{g1} \to C_{g1} &: k1 \cdot A \cdot B \\
A_{g2} \to C_{g2} &: k1 \cdot A \cdot B \\
B_{g2} \to C_{g2} &: k1 \cdot A \cdot B
\end{align*}
\noindent\texttt{inflow (none -\textgreater{} A)}
\begin{align*}
\varnothing \to A_{g1} &: b \\
\varnothing \to A_{g2} &: b
\end{align*}
\noindent\texttt{outflow (C -\textgreater{} none)}
\begin{align*}
C_{g1} \to \varnothing &: 0.01 \cdot C_{g1} \\
C_{g2} \to \varnothing &: 0.01 \cdot C_{g2}
\end{align*}
\subsubsection*{Group Stratification Transitions}
\noindent\texttt{move (g1 -\textgreater{} g2)}
\begin{align*}
A_{g1} \to A_{g2} &: 0.02 \cdot A_{g1} \\
B_{g1} \to B_{g2} &: 0.02 \cdot B_{g1} \\
C_{g1} \to C_{g2} &: 0.02 \cdot C_{g1}
\end{align*}
Total System: 6 coupled equations (3 bins $\times$ 2 group)

\subsection*{Expanded form}
\begin{align*}
\frac{A_{g1}(t+\Delta t) - A_{g1}(t)}{\Delta t} &= - (k1 \cdot A \cdot B) + (b) - (0.02 \cdot A_{g1}) \\
\frac{A_{g2}(t+\Delta t) - A_{g2}(t)}{\Delta t} &= - (k1 \cdot A \cdot B) + (b) + (0.02 \cdot A_{g1}) \\
\frac{B_{g1}(t+\Delta t) - B_{g1}(t)}{\Delta t} &= - (k1 \cdot A \cdot B) - (0.02 \cdot B_{g1}) \\
\frac{B_{g2}(t+\Delta t) - B_{g2}(t)}{\Delta t} &= - (k1 \cdot A \cdot B) + (0.02 \cdot B_{g1}) \\
\frac{C_{g1}(t+\Delta t) - C_{g1}(t)}{\Delta t} &= (2 \cdot k1 \cdot A \cdot B) - (0.01 \cdot C_{g1}) - (0.02 \cdot C_{g1}) \\
\frac{C_{g2}(t+\Delta t) - C_{g2}(t)}{\Delta t} &= (2 \cdot k1 \cdot A \cdot B) - (0.01 \cdot C_{g2}) + (0.02 \cdot C_{g1})
\end{align*}
\section*{Schedule}
Each step applies every flow once, computed from the state at the start of the step: X(n + 1) = X(n) + sum of the flows. The impulses of a step act first, and its flows run from the new state.
